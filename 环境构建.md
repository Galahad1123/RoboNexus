# CALVIN环境构建

```bash
cd RoboNexus

git clone --recurse-submodules https://github.com/mees/calvin.git
cd calvin
export CALVIN_ROOT=$(pwd)
```

```bash
conda create -n robonexus-calvin python=3.8 -y
conda activate robonexus-calvin

python -m pip install --upgrade pip
pip install setuptools==57.5.0
pip install wheel cmake==3.18.4

bash install.sh
```

出现cmake错误，修正：

```bash
python -m pip install \
  --no-cache-dir \
  --force-reinstall \
  "cmake==3.18.4.post1"

export PATH="$CONDA_PREFIX/bin:$PATH"
hash -r

echo "$CONDA_PREFIX"
type -a cmake
which cmake
cmake --version
```

修正安装脚本：

```bash
sed -i \
  's/cmake==3\.18\.4/cmake==3.18.4.post1/g' \
  install.sh

grep -n "cmake" install.sh
```

测试安装失败的依赖：

```bash
export PATH="$CONDA_PREFIX/bin:$PATH"
hash -r

python -m pip install \
  --no-cache-dir \
  --no-build-isolation \
  "MulticoreTSNE==0.1"
```

继续安装：

```bash
export PATH="$CONDA_PREFIX/bin:$PATH"
bash install.sh
```

这里默认装的 CUDA 13.0，如果你的驱动较低需要装对应版本的 CUDA

```
cp pyproject.toml pyproject.toml.bak
cp uv.lock uv.lock.bak
```

修改 `pyproject.toml`：

在文件末尾加入：

```
[tool.uv.sources]
torch = { index = "pytorch-cu128" }
torchvision = { index = "pytorch-cu128" }

[[tool.uv.index]]
name = "pytorch-cu128"
url = "https://download.pytorch.org/whl/cu128"
explicit = true
```

执行：

```bash
uv lock
uv sync
```

遇到thinplate依赖问题，该库仅在训练时使用，因此暂时移除：

```bash
cp pyproject.toml pyproject.toml.bak

python - <<'PY'
from pathlib import Path

path = Path("pyproject.toml")
text = path.read_text(encoding="utf-8")

old = "  'thinplate@git+https://github.com/cheind/py-thin-plate-spline',\n"

if old not in text:
    raise RuntimeError("没有找到 thinplate 依赖，请手动检查 pyproject.toml")

path.write_text(text.replace(old, ""), encoding="utf-8")
print("已从 pyproject.toml 移除 thinplate 依赖")
PY
```

```bash
python -m pip install -e .
```
